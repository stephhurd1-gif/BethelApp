/** * Bethel AME Lancaster — Event Flyer Feed * * Staff workflow: * 1. Create/edit the event in Google Calendar. * 2. Attach a JPG/PNG/WebP flyer/image to the event. * 3. Put ⭐ at the beginning of the title if it should be Featured. * 4. Save. * * This web app automatically finds the attached image and makes ONLY that * attached image "Anyone with the link — Viewer" so the public Events page can * display it. Non-image attachments are ignored and are NOT re-shared. * * One-time owner setup: * - Apps Script > Services (+) > add "Google Calendar API" * - Deploy > New deployment > Web app * - Execute as: Me * - Who has access: Anyone */ const BETHEL_CALENDAR_ID = 'bethelamelancaster@gmail.com'; const LOOK_AHEAD_DAYS = 370; function doGet(e) { const callback = sanitizeCallback_( e && e.parameter ? e.parameter.callback : '' ); try { const payload = buildFlyerFeed_(); const body = JSON.stringify(payload); if (callback) { return ContentService .createTextOutput(callback + '(' + body + ');') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(body) .setMimeType(ContentService.MimeType.JSON); } catch (err) { const payload = { success: false, error: String(err && err.message ? err.message : err) }; const body = JSON.stringify(payload); if (callback) { return ContentService .createTextOutput(callback + '(' + body + ');') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService .createTextOutput(body) .setMimeType(ContentService.MimeType.JSON); } } function buildFlyerFeed_() { const now = new Date(); const end = new Date( now.getTime() + LOOK_AHEAD_DAYS * 24 * 60 * 60 * 1000 ); const events = []; let pageToken = null; do { const response = Calendar.Events.list(BETHEL_CALENDAR_ID, { timeMin: now.toISOString(), timeMax: end.toISOString(), singleEvents: true, orderBy: 'startTime', maxResults: 2500, pageToken: pageToken || undefined }); (response.items || []).forEach(event => { const attachments = event.attachments || []; const image = attachments.find(isImageAttachment_); if (!image || !image.fileId) return; try { const file = DriveApp.getFileById(image.fileId); const mimeType = String(file.getMimeType() || ''); // Safety: only image files are ever made public. if (!mimeType.toLowerCase().startsWith('image/')) return; // Staff do not need to change Drive permissions manually. // The flyer becomes public because it is intentionally attached to a // public church event. try { file.setSharing( DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW ); } catch (sharingError) { console.log( 'Could not change sharing for ' + image.fileId + ': ' + sharingError ); } events.push({ iCalUID: event.iCalUID || '', title: event.summary || '', start: (event.start && (event.start.dateTime || event.start.date)) || '', imageUrl: 'https://drive.google.com/thumbnail?id=' + encodeURIComponent(image.fileId) + '&sz=w1600' }); } catch (fileError) { console.log( 'Could not read attached image ' + image.fileId + ': ' + fileError ); } }); pageToken = response.nextPageToken || null; } while (pageToken); // One image per recurring series / event UID is enough for the public page. // Keep title/start too so the homepage can safely fall back to title/date // matching if Google exposes the UID differently between feeds. const unique = {}; events.forEach(row => { if (row.iCalUID && row.imageUrl && !unique[row.iCalUID]) { unique[row.iCalUID] = row; } }); return { success: true, generatedAt: new Date().toISOString(), events: Object.values(unique) }; } function isImageAttachment_(attachment) { const mime = String(attachment.mimeType || '').toLowerCase(); if (mime.startsWith('image/')) return true; // Fallback for attachment metadata that omits MIME type. const title = String(attachment.title || '').toLowerCase(); return /\.(jpg|jpeg|png|webp|gif|heic|heif)$/i.test(title); } function sanitizeCallback_(value) { const name = String(value || '').trim(); if (!name) return ''; // JSONP callback names generated by events.html contain letters, numbers, // underscores, and dollar signs only. return /^[A-Za-z_$][A-Za-z0-9_$]*$/.test(name) ? name : ''; }
